```go
// 同步确认：关键业务
func SendSync(msg *Message) error {
    // 1. 发送消息
    result := producer.SendSync(msg)
    
    // 2. 确认结果
    if result.Status != SendOK {

    }
    
    return nil
}
```

```go
func SendAsync(msg *Message) {
```

### 1.3 事务消息

```go
// 事务消息：本地事务+消息原子性
func SendTransactionMsg(order *Order) error {
    // 1. 发送半消息
    halfMsg := producer.SendHalfMessage(msg)
    
    // 2. 执行本地事务
    tx := db.Begin()
    err := executeLocalTransaction(tx, order)
    if err != nil {
        tx.Rollback()
        producer.RollbackMessage(halfMsg)
        return err
    }
    tx.Commit()
    
    // 3. 提交消息
    producer.CommitMessage(halfMsg)
    
    return nil
}
```

### 1.4 本地消息表

```go
// 本地消息表兜底
func SendWithLocalTable(msg *Message) error {
    // 1. 写入本地消息表
    tx := db.Begin()
    err := db.Exec("INSERT INTO local_message (msg_id, status) VALUES (?, 'PENDING')", msg.ID)
    if err != nil {
        tx.Rollback()
        return err
    }
    
    // 2. 执行业务
    err = doBusiness(tx)
    if err != nil {
        tx.Rollback()
        return err
    }
    
    tx.Commit()
    
    // 3. 异步发送
    go func() {
        err := producer.Send(msg)
        if err != nil {
            // 发送失败，定时重试
            retrySend(msg)
        } else {
            // 更新状态
            db.Exec("UPDATE local_message SET status = 'SENT' WHERE msg_id = ?", msg.ID)
        }
    }()
    
    return nil
}

// 定时扫描补发
func RetryUnsentMessages() {
    ticker := time.NewTicker(5 * time.Minute)
    for range ticker.C {
        unsentMsgs := db.Query("SELECT * FROM local_message WHERE status = 'PENDING'")
        for _, msg := range unsentMsgs {
            retrySend(msg)
        }
    }
}
```

## 二、Broker端可信

### 2.1 同步刷盘

```
同步刷盘：
- 消息写入磁盘后才返回成功
- 牺牲性能换绝对可靠
- 适合金融级业务

异步刷盘：
- 消息写入内存即返回
- 后台异步刷盘
- 兼顾性能和可靠性
- 通用场景首选
```

### 2.2 主从同步

```
同步复制：
- 主节点写入成功后复制到从节点
- 主节点宕机时从节点有完整副本
- 适合高可靠场景

异步复制：
- 主节点写入后异步复制
- 性能更好，但可能丢消息
- 适合高性能场景
```

### 2.3 死信队列

```go
// 处理消费失败消息
func HandleFailedMessage(msg *Message) {
    // 1. 超过重试次数
    if msg.RetryCount >= MaxRetry {
        // 2. 进入死信队列
        deadLetterQueue.Send(msg)
        
        // 3. 告警
        alert("消息进入死信队列: %v", msg.ID)
    }
}

// 定期清理过期消息
func CleanDeadLetterQueue() {
    ticker := time.NewTicker(1 * time.Hour)
    for range ticker.C {
        expiredMsgs := deadLetterQueue.GetExpired()
        for _, msg := range expiredMsgs {
            deadLetterQueue.Delete(msg)
        }
    }
}
```

## 三、消费端可信

### 3.1 手动ACK

```go
// 手动ACK：业务成功后确认
func ConsumeMessage(msg *Message) error {
    // 1. 执行业务
    err := doBusiness(msg)
    if err != nil {
        // 业务失败，不ACK，触发重试
        return err
    }
    
    // 2. 业务成功，手动ACK
    consumer.ACK(msg)
    
    return nil
}
```

### 3.2 幂等设计

```go
// 幂等性设计：唯一消息ID去重
func ConsumeWithIdempotent(msg *Message) error {
    // 1. 查询是否已消费
    record := db.GetConsumeRecord(msg.ID)
    if record != nil {
        return nil // 已消费，幂等返回
    }
    
    // 2. 执行业务
    err := doBusiness(msg)
    if err != nil {
        return err
    }
    
    // 3. 记录消费记录
    db.InsertConsumeRecord(msg.ID)
    
    return nil
}
```

### 3.3 批量消费

```go
// 批量消费：逐条处理异常
func ConsumeBatch(messages []*Message) {
    for _, msg := range messages {
        err := consumeSingle(msg)
        if err != nil {
            // 单条失败不影响整批
            handleFailedMessage(msg)
        }
    }
}
```

```go
    if !startTime.IsZero() {
        messages := consumer.ReplayByTime(startTime)
```
